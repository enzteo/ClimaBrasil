import * as cheerio from "cheerio";
import { prisma } from "./prisma";
import { fetchRobotsRules, isPathAllowed } from "./robots";

const USER_AGENT = "MeuSEOBot/1.0";
const DELAY_MS = 500; // 2 requisições por segundo = 1 a cada 500ms
const MAX_PAGES = 200;

function sleep(ms: number) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

// Normaliza uma URL: remove barra final duplicada, mantém só o essencial.
function normalizeUrl(url: string): string {
  try {
    const u = new URL(url);
    u.hash = ""; // ignora âncoras (#secao)
    let path = u.pathname;
    if (path.length > 1 && path.endsWith("/")) {
      path = path.slice(0, -1);
    }
    return `${u.origin}${path}${u.search}`;
  } catch {
    return url;
  }
}

// Tenta ler o sitemap.xml e extrair as URLs listadas nele (regex simples, sem lib de XML).
async function fetchSitemapUrls(baseUrl: string): Promise<string[]> {
  try {
    const response = await fetch(`${baseUrl}/sitemap.xml`, {
      headers: { "User-Agent": USER_AGENT },
    });
    if (!response.ok) return [];
    const xml = await response.text();
    const matches = [...xml.matchAll(/<loc>(.*?)<\/loc>/g)];
    return matches.map((m) => m[1].trim());
  } catch {
    return [];
  }
}

export async function crawlDomain(auditRunId: string, hostname: string) {
  const baseUrl = `https://${hostname}`;

  await prisma.auditRun.update({
    where: { id: auditRunId },
    data: { status: "CRAWLING" },
  });

  const { disallowedPaths } = await fetchRobotsRules(baseUrl, USER_AGENT);
  const sitemapUrls = await fetchSitemapUrls(baseUrl);

  const visited = new Set<string>();
  const queue: string[] = [normalizeUrl(baseUrl), ...sitemapUrls.map(normalizeUrl)];
  // Guarda, pra cada URL visitada, quantos links internos apontam pra ela.
  const incomingLinksCount = new Map<string, number>();

  let pagesCrawled = 0;

  while (queue.length > 0 && pagesCrawled < MAX_PAGES) {
    const currentUrl = queue.shift()!;
    if (visited.has(currentUrl)) continue;

    const urlObj = new URL(currentUrl);
    if (urlObj.hostname !== hostname) continue; // só rastreia o próprio domínio
    if (!isPathAllowed(urlObj.pathname, disallowedPaths)) continue;

    visited.add(currentUrl);

    const startTime = Date.now();
    let statusCode: number | null = null;
    let redirectedTo: string | null = null;
    let html = "";

    try {
      const response = await fetch(currentUrl, {
        headers: { "User-Agent": USER_AGENT },
        redirect: "follow",
      });
      statusCode = response.status;
      if (response.redirected && response.url !== currentUrl) {
        redirectedTo = response.url;
      }
      html = await response.text();
    } catch {
      statusCode = 0; // erro de rede (site fora do ar, DNS, etc.)
    }

    const responseTimeMs = Date.now() - startTime;

    let title: string | null = null;
    let metaDescription: string | null = null;
    let h1Count = 0;
    let wordCount = 0;
    let canonicalUrl: string | null = null;
    let hasStructuredData = false;
    let hasViewportMeta = false;

    if (html) {
      const $ = cheerio.load(html);

      title = $("title").first().text().trim() || null;
      metaDescription =
        $('meta[name="description"]').attr("content")?.trim() || null;
      h1Count = $("h1").length;
      canonicalUrl = $('link[rel="canonical"]').attr("href") || null;
      hasStructuredData = $('script[type="application/ld+json"]').length > 0;
      hasViewportMeta = $('meta[name="viewport"]').length > 0;

      const bodyText = $("body").text().replace(/\s+/g, " ").trim();
      wordCount = bodyText ? bodyText.split(" ").length : 0;

      // Extrai todos os links da página e enfileira os internos ainda não visitados
      $("a[href]").each((_, el) => {
        const href = $(el).attr("href");
        if (!href) return;

        let absoluteUrl: string;
        try {
          absoluteUrl = new URL(href, currentUrl).toString();
        } catch {
          return;
        }

        const normalized = normalizeUrl(absoluteUrl);
        let isInternal = false;

        try {
          isInternal = new URL(normalized).hostname === hostname;
        } catch {
          isInternal = false;
        }

        if (isInternal) {
          incomingLinksCount.set(
            normalized,
            (incomingLinksCount.get(normalized) || 0) + 1
          );
          if (!visited.has(normalized) && !queue.includes(normalized)) {
            queue.push(normalized);
          }
        }
      });
    }

    await prisma.page.create({
      data: {
        auditRunId,
        url: currentUrl,
        statusCode,
        redirectedTo,
        responseTimeMs,
        title,
        metaDescription,
        h1Count,
        wordCount,
        canonicalUrl,
        hasStructuredData,
        hasViewportMeta,
        html,
      },
    });

    pagesCrawled++;

    await prisma.auditRun.update({
      where: { id: auditRunId },
      data: { pagesCrawled },
    });

    await sleep(DELAY_MS);
  }

  // Depois de rastrear tudo, atualiza quantos links internos cada página recebeu
  // (é assim que detectamos página órfã: incomingLinks === 0)
  const pages = await prisma.page.findMany({ where: { auditRunId } });
  for (const page of pages) {
    const count = incomingLinksCount.get(normalizeUrl(page.url)) || 0;
    await prisma.page.update({
      where: { id: page.id },
      data: { incomingLinks: count },
    });
  }

  await prisma.auditRun.update({
    where: { id: auditRunId },
    data: { status: "ANALYZING" },
  });

  return pagesCrawled;
}

Reparei um detalhe importante: o arquivo está como Crawler.ts (com C maiúsculo), mas o correto é crawler.ts (tudo minúsculo) — isso é crítico porque:

No Windows, maiúscula/minúscula não faz diferença pro sistema de arquivos, então localmente vai parecer que funciona.
Mas o container Docker roda Linux (a imagem node:20-alpine), e no Linux nome de arquivo é case-sensitive — Crawler.ts e crawler.ts são dois arquivos diferentes. Quando o código tentar import { crawlDomain } from "./crawler" (tudo minúsculo, como vamos escrever), vai funcionar no seu PC mas quebrar no Docker/no servidor da empresa.
Renomeia agora, antes de esquecer: clica com botão direito no arquivo no VS Code → Renomear → crawler.ts (minúsculo).

O que cada rota faz:

POST /api/audits — você manda { "hostname": "algumsite.com" }, ela cria o registro do domínio (ou reaproveita se já existir, com upsert), cria uma nova execução de auditoria (AuditRun), e dispara o crawler sem esperar ele terminar (repara que não tem await antes de crawlDomain(...) — só um .catch() pra capturar erro). Isso é o que cumpre a exigência "roda em segundo plano, a tela não fica travada".
GET /api/audits/{id} — consulta em que pé está uma auditoria específica (quantas páginas já rastreou, se terminou, etc.). A tela vai chamar essa rota de tempos em tempos pra atualizar a barra de progresso.
